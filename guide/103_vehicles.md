🏷️ ***__КЛЕЙМ ТЕХНИКИ__***
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Технику ***клеймят*** до того как сел за руль. Либо называешь сквад `«Танк / Верт / ПТУР»` (неймклейм). Без клейма придёт другой сквад и скажет что ты угнал их технику.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
👤 ***__КТО ВОДИТ__***
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

>>> • ***Мехвод / крювмен*** — наземная техника. Если не мехвод — через 5 сек выкинет.
• ***Пилот*** — вертолёты.
• ***Чинилка / ремонтка*** — если техника дымит, вези на ремонт.
• ***РВС (RWS)*** — дистанционное управление оружием. «Закрытая техника» — стрельба изнутри.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
📦 ***__ЛОГИСТИКА (СОПЛЯ / СУПЛАЙКА)__***
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

>>> ***Главная машина на поле боя. Без неё не будет фобы.***

• 1 сопля = 2 ящика × 50 матов = `100` ресов
• Выгруз — кнопка `X`
• Загрузка — авто пополнение на мэйне
• Ресы тратятся на: постройки, респавн (10 матов), пополнение БК (10 матов)
• ***~~Если соплю слили — команда без ресов~~***
• ***Не уводи соплю покататься*** — `#сквадной` скажет спасибо матом
> У повстанцев и ополченцев сопли бывают гусеничные — едет очень долго.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
👥 ***__ЭКИПАЖ ТЕХНИКИ__***
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

• Водитель смотрит дорогу. Наводчик ищет цели.
• ***Один в технике*** — ты лёгкая цель для ЛАТ / трубы или ХАТ / танкема.
• После боя — отгони технику в укрытие чиниться.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
💸 ***__ПОТЕРИ__***
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

>>> • Слил технику — штраф по тикетам. Чем тяжелее, тем больнее.
• ***Не бросай технику.*** Если горит — выпрыгивай.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
💡 ***__ПОЛЕЗНОЕ__***
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

• ***Труба / тандем*** — главная угроза. Если по тебе работают — маневрируй.
• На ***мэйне*** техника чинится и заправляется бесплатно.
• ***Закрытая фоба*** — в здании (защищена от миномётов / авиации).
• ***Открытая фоба*** — на земле (уязвима).

Сленг по технике — в `#словарь`.
